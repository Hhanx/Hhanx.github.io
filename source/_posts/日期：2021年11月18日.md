### TCP端口扫描：

## 首先：

因为是winsows底下写代码所以每次创建一个socket之前都需要初始化以下版本型号；

```c++
   如：
	WSADATA wsdata;                        //对应的套接网规格的版本
    WSAStartup(MEKEWORD(2,2),&wsdata);     //需要注意的是第二个参数是个指针，所以需要获取地址
```

## 其次：

其次就是需要定义一个**套接字**，使用**for循环**一直**连接**指定的**端口**

```c++
如：
    int startPort,endPort;
    for (int currPort = startPort; currPort <= endPort; currPort++)           //从端口startPort一直循环到endPort
	{
		sock = socket(PF_INET, SOCK_STREAM, IPPROTO_TCP);                    //创建套接字
		sock_Addr.sin_port = htons(currPort);                                
		int ret = connect(sock, (sockaddr*)&sock_Addr, sizeof(sock_Addr));   //连接函数，成功为0，不成功为非0
		if (ret == SOCKET_ERROR)
		{
			cout << "Success" << "   " << lp <<"          " << currPort << endl;
			closesocket(sock);
		}
		else
		{
			cout << "Filled " << "   " << lp << "          " << currPort << endl;
		}
	}


```

### 字符串处理:

```c++
string Port;
字符串的长度：Port.length(),返回的就是长度了//直接返回长度
字符串的转换：把string转化成char*函数就是这样：Port.c_str() //直接返回char*字符
字符串的截取：Port.substr(参数一，参数二)//参数1是截取首的长度，参数2是截取尾的长度
字符串的字符查找：Port.find(参数一，参数二);//参数1是一个字符，参数2是查找的起始位置
字符串转换成数字： atoi//直接转   
    
```

## 所以：

### 整体代码大概就是这样：

```c++
#define _WINSOCK_DEPRECATED_NO_WARNINGS
#include<iostream>
#include<conio.h>
#include<stdlib.h>
#include<string>
#include<WinSock2.h>
#pragma comment(lib,"ws2_32.lib")
using namespace std;
int scant(const char* lp, int startPort, int endPort);
int main()
{
	char lp[32] = "";
	char* plp = lp;
	string Port;
	int startPort = 0,endPort = 0;
	cout << "请输入想要检测的IP地址：" << endl;
	cin >> lp;
	cout << "请输入想要检测的端口范围:(例如：100-1000;用“-”隔开)" << endl;
	cout << "当程序运行中时，按一次空格结束程序！" << endl;
	cin >> Port;
	int po_em = Port.find('-',0);//‘-’的位置
	startPort = atoi(Port.substr(0, po_em).c_str());//截取字符串
	endPort = atoi(Port.substr(po_em+1, Port.length()).c_str());

	scant(lp,startPort,endPort);

}
int scant(const char* lp, int startPort, int endPort)
{
	WSADATA wsData;
	SOCKET sock;
	sockaddr_in sock_Addr;
	WSAStartup(MAKEWORD(2, 2), &wsData);
	sock_Addr.sin_family = AF_INET;
	sock_Addr.sin_addr.S_un.S_addr = inet_addr(lp);
	cout << "CLASS" << "        IP        " << "        PORT        " << endl;
	for (int currPort = startPort; currPort <= endPort; currPort++)
	{
		if (_kbhit() && _getch() == ' ')
		{
			break;
		}
		sock = socket(PF_INET, SOCK_STREAM, IPPROTO_TCP);
		sock_Addr.sin_port = htons(currPort);
		int ret = connect(sock, (sockaddr*)&sock_Addr, sizeof(sock_Addr));
		if (ret == SOCKET_ERROR)
		{
			cout << "Success" << "   " << lp <<"          " << currPort << endl;
			closesocket(sock);
		}
		else
		{
			cout << "Filled " << "   " << lp << "          " << currPort << endl;
		}
	}
	WSACleanup();
	return 0;
}
```

问题测试/解决方法（分析问题引发的可能性及相应的测试和解决办法，注明解决结果）：

这个问题就是编译器识别不到main函数，导致程序出错使其不完整；

### 解决办法：

1.菜单中选择 Project->Properties, 弹出Property Pages窗口

2.在左边栏中依次选择：Configuration Properties->C/C++->Preprocessor,然后在右边栏的Preprocessor Definitions对应的项中删除_WINDOWS, 添加_CONSOLE.

3.在左边栏中依次选择：Configuration Properties->Linker->System,然后在右边栏的SubSystem对应的项改为CONSOLE(/SUBSYSTEM:CONSOLE)
