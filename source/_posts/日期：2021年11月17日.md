```c++
//动态库里面
// dllmain.cpp : 定义 DLL 应用程序的入口点。
#include "pch.h"

extern "C" __declspec(dllexport) void aoligei();      // 这个是导出函数，把函数/变量导
void aoligei()
{
    MessageBox(NULL,L"恭喜你打开成功",L"提示框", MB_OKCANCEL);
}

BOOL APIENTRY DllMain( HMODULE hModule,
                       DWORD  ul_reason_for_call,
                       LPVOID lpReserved
                     )
{
    switch (ul_reason_for_call)
    {
    case DLL_PROCESS_ATTACH:
    case DLL_THREAD_ATTACH:
    case DLL_THREAD_DETACH:
    case DLL_PROCESS_DETACH:
        break;
    }
    return TRUE;
}
```

```c++
//外部代码
#include<windows.h>
typedef void(*Message_Box)();
int main()
{
	HMODULE hModule = LoadLibrary(L"firstddl.dll");      //把动态库导入到内存空间中
    if(hModule == NULL){                                 //如果导入成功就返回非0，否者NULL
        return -1;
    }
    else{
        MessageBox(NULL,L"firstdll加载成功", L"提示:",MB_OK);
    }
	Message_Box Msg = GetProcAddress(hModule,L"aoligei");       //把库中的函数检索出变量或者函数的地址，如果失败返回NULL
    
    Msg();
    
	return 0;
}

```
