BOOL CXDDlg::OnEraseBkgnd(CDC* pDC)
{
	

	BOOL bRet =  CDialogEx::OnEraseBkgnd(pDC);
	int nRed, nGreen, nBluel;
	nRed = m_check1_red.GetCheck() ? 255 : 0;
	nGreen = m_check2_lu.GetCheck() ? 255 : 0;
	nBluel = m_check3_blue.GetCheck() ? 255 : 0;
	//设置颜色
	COLORREF color = RGB(nRed,nGreen,nBluel);
	//创建画刷
	CBrush brush;
	//存入颜色
	brush.CreateSolidBrush(color);
	//保存旧的刷子
	CBrush* poldBrush = pDC->SelectObject(&brush);
	//画的大小以及坐标
	RECT rc = { 20,20,100,100 };
	//判断是否点击复选框
	//若是就是话矩形，否者圆形
	if (m_radio1.GetCheck())
	{
		pDC->Rectangle(&rc);
	}
	else
	{
		pDC->Ellipse(&rc);
	}
	//放入新创建的刷子
	pDC->SelectObject(poldBrush);

	return bRet;
}
