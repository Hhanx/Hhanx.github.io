首先要记住combo，list空间的基本处理函数

.AddString() //这个函数就是写入字符串

.GetCount()//这个函数就是获取字符串的总个数

.GetCurSel()//这个函数就是获取当时点击的的条件来过去当时的长度

.DeleteString()//这个就是删除函数

```c++
void CMFCScrollBarDlg::OnBnClickedButton1()
{
	CString str;
	GetDlgItemText(IDC_EDIT1,str);
	m_combo.AddString(str);
	CString strlen;
	int len = str.GetLength();
	strlen.Format(L"长度%d，第%d行",len,m_combo.GetCount());
	m_list.AddString(str);
	m_list.AddString(strlen);
}
void CMFCScrollBarDlg::OnBnClickedButton2()
{
	int nIndex = m_combo.GetCurSel();
	m_list.DeleteString(nIndex);
}
```
