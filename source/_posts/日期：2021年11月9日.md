//矩形数组是就是  [行，列]//通常用在图形当中
1.矩形数组的初始化方法
	1.int[,] rectangle = new int[,] {{1,2,...}，{3,4,...}，{5,6,...}，...};//在大括号里面包裹着的一个个小括号是行，每一行有多个数，这个数就是对应的列

	2.int[,] rectangle = new int[2,2];
	  for(int i = 0; i < rectangle.GetLength(0)); i++)
      {
          for(int j = 0; i < rectangle.GetLength(1);j++)
          {
              rectangle[i,j] = 100; // 给所有的数都赋值为100
          }
      }

2.锯齿数组的初始化方法
    1.int[][] sawtooth = new int[6][];//这种初始化方法需要手动输入，可以用for循环去输入
	
	2.int[][] sawtooth = new int[][]{

    };

###### 2.接受参数

```c#
//接受参数我的理解  就是像个数组一样的用法
比如：
    int a,b,c;
    Conselo.Write("{}{}{}",a,b,c);
//引号右边的变量都是值，有多少个值对应的就是下标几 如果有3个变量那么最大就是{2}，一起就是{0}{1}{2}；
//在右边的值可以不按顺序写,但是要注意的就是左边的“{}”一定要按照原有的和变量数量规定下标
```
