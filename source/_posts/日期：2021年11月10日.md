###### 2.接受参数/传参

```c#
1.传参
    class mymain
    {
        static void mytext(int a)
        {
            a = a + 1;
            Console.WriteLine(a); //序号一
        }
        static void Main()
        {
            int a = 100;
            mymain(a);
            Console.WriteLine(a); //序号二
        }
    }
    此时序号一输出为：101    
    此时序号二输出为：100
2.修饰符号传参(ref)
    class mymain
    {
        static void mytext(ref int a)
        {
            a = a + 1;
            Console.WriteLine(a); //序号一
        }
        static void Main()
        {
            ref int a = 100;
            mymain(a);
            Console.WriteLine(a); //序号二
        }
    }
    此时序号一输出为：101    
    此时序号二输出为：101
 3.修饰符号接收参数（out）
   //用在函数调用返回参数
    class mymain
    {
        static void mytext(int a,out int b,out int c)
        {
           	 b = a+100;//序号一
             c = b+100;
        }
        static void Main()
        {
            int a,b;
			mymain(100,out a,out b);
            Console.WriteLine("a={0},b={1}",a,b);
           //a = 200;
           //b = 300;
        }
    }   
//out修饰符号 就是用来接受参数的
    此时序号一输出为：  200
    此时序号二输出为：  300
 4.修饰符号数组传参参（params） 
        static void Main(String[] args)
   		{
         int sum =  mytext(1,2,3,4,5);//从main函数里面传递参数到自定义函数
            Console.WriteLine(sum);

        }
        static int mytext(params int[] arr)
        {
            int sum = 0;
            for(int i = 0;i<arr.Length;i++)
            {
                sum += arr[i];
            }
            return sum; //运行完了就返回
        }   
   5.用名字识别参数   
        static void Main(String[] args)
   		{
			mytext(c:100);//指定一个名字赋值
        }
        static void mytext(int a = 0,int b = 0,int c = 0)//或者 static void mytext(int a,int b,int c)
        {
            Console.WriteLine(c);//c = 100
        }
        
```

###### 3.隐式类型

```c#
如果觉得给值比较累，可以用静态的var来定义
var add = "shuaige";     
等价于：   
string add = "shuaige"；  
因为是静态的不能分开赋值如:    
var add；
add = "shuaige";
这是错误的写法
```
